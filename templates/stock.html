```python # 股票查詢功能 @app.route('/stock', methods=['GET', 'POST']) def stock(): if request.method == 'POST': # 讀取使用者輸入的股票代號 question = request.form.get('question', '').strip() # 設定初始答案 answer = "" # API URL url = f"https://www.twse.com.tw/exchangeReport/STOCK_DAY?response=json&stockNo={question}" try: # 發送請求 res = requests.get(url, timeout=10) data = res.json() # 判斷是否成功 if data["stat"] == "OK" and data["data"]: answer = data["data"][-1][6] else: answer = "查無資料，請確認股票代號。" except Exception: answer = "查詢失敗，請稍後再試。" # 回傳答案 return render_template( 'stock.html', question=question, answer=answer ) # GET 時給空白欄位 return render_template( 'stock.html', question="", answer="" ) ```
